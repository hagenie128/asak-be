UserMenuMapper

UserMenuMapper.java
package com.asak.user.mapper;

import java.util.List;
import org.apache.ibatis.annotations.Param;
import com.asak.admin.dto.response.MenuOptPolicyResponse;
import com.asak.user.dto.response.CategoryResponse;
import com.asak.user.dto.response.MenuDetailResponse;
import com.asak.user.dto.response.MenuListResponse;

public interface UserMenuMapper {
  List<CategoryResponse> getCategories();
  List<MenuListResponse> getMenuList(@Param("categoryId") Long categoryId);
  MenuDetailResponse getMenuDetail(@Param("menuId") Long menuId);
  List<MenuOptPolicyResponse> getMenuOptions(@Param("menuId") Long menuId);
}
UserMenuMapper.xml

옵션은 Admin의 MenuOptPolicyResponse를 재사용 — 필드가 이미 Kiosk 계약 그대로라서요.

<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE mapper PUBLIC "-//mybatis.org//DTD Mapper 3.0//EN" "https://mybatis.org/dtd/mybatis-3-mapper.dtd">
<mapper namespace="com.asak.user.mapper.UserMenuMapper">

  <select id="getCategories" resultType="com.asak.user.dto.response.CategoryResponse">
    SELECT id AS categoryId, name, sort_no AS sortOrder
    FROM category
    WHERE active = 1
    ORDER BY sort_no
  </select>

  <select id="getMenuList" resultType="com.asak.user.dto.response.MenuListResponse">
    SELECT menu_id AS menuId, category_id AS categoryId, name, price, image_url AS imageUrl,
           base_kcal AS baseKcal, is_sold_out AS isSoldOut, has_sold_out_ingredient AS hasSoldOutIngredient
    FROM vw_menu_list
    WHERE category_id = #{categoryId}
  </select>

  <resultMap id="menuDetailMap" type="com.asak.user.dto.response.MenuDetailResponse">
    <id column="menu_id" property="menuId"/>
    <result column="category_id" property="categoryId"/>
    <result column="name" property="name"/>
    <result column="price" property="price"/>
    <result column="image_url" property="imageUrl"/>
    <result column="description" property="description"/>
    <result column="base_kcal" property="baseKcal"/>
    <result column="is_sold_out" property="isSoldOut"/>
    <result column="has_sold_out_ingredient" property="hasSoldOutIngredient"/>
    <result column="is_orderable" property="isOrderable"/>
    <result column="allergy_text" property="allergyText"/>
    <result column="allergens" property="allergens"/>
    <collection property="ingredients" ofType="com.asak.user.dto.response.IngredientResponse"
                column="menu_id" select="findKioskIngredients"/>
  </resultMap>

  <select id="getMenuDetail" resultMap="menuDetailMap">
    SELECT m.id AS menu_id, m.cat_id AS category_id, m.name, m.price, m.image_url,
           m.description, mn.kcal AS base_kcal,
           va.direct_sold_out AS is_sold_out,
           (va.has_core_sold_out OR va.base_ing_exhausted OR va.base_opt_exhausted OR va.has_blocking_standard) AS has_sold_out_ingredient,
           NOT (va.direct_sold_out OR va.has_core_sold_out OR va.base_ing_exhausted OR va.base_opt_exhausted
                OR va.has_blocking_standard OR va.has_exhausted_required_group) AS is_orderable,
           COALESCE(ag.allergy_text, '') AS allergy_text,
           COALESCE(ag.allergens, JSON_ARRAY()) AS allergens
    FROM menu m
    JOIN vw_menu_availability va ON va.menu_id = m.id
    LEFT JOIN menu_nutr mn ON mn.menu_id = m.id
    LEFT JOIN (
        SELECT menu_id,
               JSON_ARRAYAGG(name) AS allergens,
               GROUP_CONCAT(name ORDER BY name SEPARATOR ', ') AS allergy_text
        FROM (
            SELECT DISTINCT mi.menu_id, a.name
            FROM menu_ing mi
            JOIN ing_allergen ia ON ia.ing_id = mi.ing_id
            JOIN allergen a ON a.id = ia.allergen_id
            UNION
            SELECT DISTINCT mop.menu_id, a.name
            FROM menu_opt_policy mop
            JOIN opt_policy_item opi ON opi.policy_id = mop.policy_id
            JOIN opt_item oi ON oi.id = opi.opt_item_id
            JOIN ing_allergen ia ON ia.ing_id = oi.ing_id
            JOIN allergen a ON a.id = ia.allergen_id
        ) all_allerg
        GROUP BY menu_id
    ) ag ON ag.menu_id = m.id
    WHERE m.id = #{menuId}
  </select>

  <select id="findKioskIngredients" resultType="com.asak.user.dto.response.IngredientResponse">
    SELECT DISTINCT ingredient_id AS ingredientId, ing_name AS name, can_remove AS canRemove, ing_sold_out AS isSoldOut
    FROM vw_menu_ing_json
    WHERE menu_id = #{menuId}
  </select>

  <select id="getMenuOptions" resultType="com.asak.admin.dto.response.MenuOptPolicyResponse">
    SELECT option_group_id AS optionGroupId, name, group_type AS groupType, select_type AS selectType,
           min_select AS minSelect, max_select AS maxSelect, sort_order AS sortOrder,
           is_required AS isRequired, items
    FROM vw_menu_opt_policy_json
    WHERE menu_id = #{menuId}
  </select>

</mapper>

UserOrderMapper

주문 생성 전용. 결제 관련 메서드는 UserPayMapper로 분리했습니다 — 실제 프로젝트에 UserOrderController(orderService)와 UserPayController(payService)가 이미 따로 있어서요.

UserOrderMapper.java
package com.asak.user.mapper;

import java.util.List;
import org.apache.ibatis.annotations.Param;
import com.asak.user.mapper.dto.MenuPriceCheck;
import com.asak.user.mapper.dto.OptItemPriceCheck;
import com.asak.user.mapper.dto.OrderInsertParams;
import com.asak.user.mapper.dto.OrderItemInsertParams;

public interface UserOrderMapper {
  MenuPriceCheck findMenuPriceCheck(@Param("menuId") Long menuId);
  List<OptItemPriceCheck> findOptionItemPriceChecks(@Param("ids") List<Long> ids);
  String generateOrderNo();

  void insertOrder(OrderInsertParams params);
  void insertOrderItem(OrderItemInsertParams params);
  void insertOrderItemOptions(@Param("orderItemId") Long orderItemId,
                               @Param("optionItemIds") List<Long> optionItemIds,
                               @Param("prices") List<Integer> prices);
  void insertItemExclusions(@Param("orderItemId") Long orderItemId,

UserOrderMapper.xml
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE mapper PUBLIC "-//mybatis.org//DTD Mapper 3.0//EN" "https://mybatis.org/dtd/mybatis-3-mapper.dtd">
<mapper namespace="com.asak.user.mapper.UserOrderMapper">

  <select id="findMenuPriceCheck" resultType="com.asak.user.mapper.dto.MenuPriceCheck">
    SELECT m.price,
           NOT (va.direct_sold_out OR va.has_core_sold_out OR va.base_ing_exhausted OR va.base_opt_exhausted
                OR va.has_blocking_standard OR va.has_exhausted_required_group) AS orderable
    FROM menu m JOIN vw_menu_availability va ON va.menu_id = m.id
    WHERE m.id = #{menuId}
  </select>

  <select id="findOptionItemPriceChecks" resultType="com.asak.user.mapper.dto.OptItemPriceCheck">
    SELECT id AS optionItemId, add_price AS addPrice, sold_out AS soldOut
    FROM opt_item
    WHERE id IN
    <foreach item="id" collection="ids" open="(" separator="," close=")">#{id}</foreach>
  </select>

  <select id="generateOrderNo" resultType="string">
    SELECT CONCAT('ASAK-', DATE_FORMAT(NOW(), '%Y%m%d'), '-',
                  LPAD((SELECT COUNT(*) + 1 FROM orders WHERE DATE(created_at) = CURDATE()), 3, '0'))
  </select>

  <insert id="insertOrder" useGeneratedKeys="true" keyProperty="id">
    INSERT INTO orders (order_no, order_type_id, status_id, total_price, created_at)
    VALUES (
      #{orderNo},
      (SELECT id FROM common_code WHERE code = #{orderType}
         AND code_grp_id = (SELECT id FROM code_group WHERE group_code = 'ORDER_TYPE')),
      (SELECT id FROM common_code WHERE code = 'RECEIVED'
         AND code_grp_id = (SELECT id FROM code_group WHERE group_code = 'ORDER_STATUS')),
      #{totalAmount}, NOW()
    )
  </insert>

  <insert id="insertOrderItem" useGeneratedKeys="true" keyProperty="id">
    INSERT INTO order_item (order_id, menu_id, quantity, price)
    VALUES (#{orderId}, #{menuId}, #{quantity}, #{unitPrice})
  </insert>

  <insert id="insertOrderItemOptions">
    INSERT INTO order_item_option (order_item_id, opt_item_id, quantity, price)
    VALUES
    <foreach item="optId" collection="optionItemIds" index="i" separator=",">
      (#{orderItemId}, #{optId}, 1, #{prices[${i}]})
    </foreach>
  </insert>

  <insert id="insertItemExclusions">
    INSERT INTO item_exclusion (order_item_id, ing_id)
    VALUES
    <foreach item="ingId" collection="ingredientIds" separator=",">
      (#{orderItemId}, #{ingId})
    </foreach>
  </insert>

</mapper>

UserPayMapper

결제 승인·결제수단 전용. UserPayController(payService)가 쓸 매퍼입니다.

UserPayMapper.java
package com.asak.user.mapper;

import java.util.List;
import org.apache.ibatis.annotations.Param;
import com.asak.user.dto.response.PaymentMethodResponse;
import com.asak.user.dto.response.PaymentResultResponse;
import com.asak.user.mapper.dto.OrderPaymentCheck;
import com.asak.user.mapper.dto.PaymentInsertParams;

public interface UserPayMapper {
  List<PaymentMethodResponse> getPaymentMethods();
  PaymentResultResponse getPaymentResult(@Param("paymentId") Long paymentId);

  OrderPaymentCheck findOrderForPayment(@Param("orderId") Long orderId);
  boolean isPaymentMethodEnabled(@Param("methodCode") String methodCode);
  void insertPayment(PaymentInsertParams params);
}
UserPayMapper.xml
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE mapper PUBLIC "-//mybatis.org//DTD Mapper 3.0//EN" "https://mybatis.org/dtd/mybatis-3-mapper.dtd">
<mapper namespace="com.asak.user.mapper.UserPayMapper">

  <select id="getPaymentMethods" resultType="com.asak.user.dto.response.PaymentMethodResponse">
    SELECT c.code AS methodCode, pm.name AS methodName, pm.active AS isEnabled, pm.sort_no AS sortOrder
    FROM pay_method_cfg pm
    JOIN common_code c ON c.id = pm.method_id
    ORDER BY pm.sort_no
  </select>

  <select id="getPaymentResult" resultType="com.asak.user.dto.response.PaymentResultResponse">
    SELECT payment_id AS paymentId, order_id AS orderId, order_no AS orderNo,
           payment_status AS paymentStatus, approved_amount AS approvedAmount,
           approved_at AS approvedAt, waiting_order_count AS waitingOrderCount
    FROM vw_payment_result
    WHERE payment_id = #{paymentId}
  </select>

  <!-- 결제 승인용: 주문 총액 + 기존 결제상태(있으면) 조회 -->
  <select id="findOrderForPayment" resultType="com.asak.user.mapper.dto.OrderPaymentCheck">
    SELECT o.total_price AS totalPrice, ps.code AS existingPaymentStatus
    FROM orders o
    LEFT JOIN payment p ON p.order_id = o.id
    LEFT JOIN common_code ps ON ps.id = p.status_id
    WHERE o.id = #{orderId}
  </select>

  <select id="isPaymentMethodEnabled" resultType="boolean">
    SELECT pm.active
    FROM pay_method_cfg pm
    JOIN common_code c ON c.id = pm.method_id
    WHERE c.code = #{methodCode}
  </select>

  <!-- id를 담을 파라미터 객체(PaymentInsertParams)로 받음 — @Param 여러 개로는 useGeneratedKeys가 값을 못 받음 -->
  <insert id="insertPayment" useGeneratedKeys="true" keyProperty="id">
    INSERT INTO payment (order_id, method_id, status_id, amount, paid_at)
    VALUES (
      #{orderId},
      (SELECT id FROM common_code WHERE code = #{methodCode}
         AND code_grp_id = (SELECT id FROM code_group WHERE group_code = 'PAYMENT_METHOD')),
      (SELECT id FROM common_code WHERE code = 'APPROVED'
         AND code_grp_id = (SELECT id FROM code_group WHERE group_code = 'PAYMENT_STATUS')),
      #{amount}, NOW()
    )
  </insert>

</mapper>

Service 힌트 (코드 없음)

Controller·Service 코드는 여기 안 씁니다 — 뭘 조심해야 하는지만. 실제 코드 짤 사람이 UserOrderService/UserPayService 각각에 나눠 넣으면 됩니다.

힌트

  • 가격은 절대 클라이언트 값 신뢰 안 함 — 메뉴 현재가/옵션 현재가로 서버가 재계산
  • 저장 직전 품절 여부 재확인 — 요청 시점과 저장 시점 사이에 품절될 수 있음
  • 주문번호는 사전 락 안 걸고, UNIQUE 제약 걸어둔 채 충돌나면 재시도
  • 품목+옵션+제외재료는 한 트랜잭션 — 하나라도 실패하면 전부 롤백

힌트 — UserOrderService.createOrder (Controller/Service 코드는 안 씀, 순서만)

  • 품목별로 findMenuPriceCheck/findOptionItemPriceChecks 호출 → 가격·품절 재검증 (클라이언트 값 신뢰 안 함)
  • generateOrderNo + insertOrder를 반복문으로 감싸서, DuplicateKeyException 잡히면 새 번호로 재시도
  • 품목마다 insertOrderItem → 받은 orderItemId로 옵션/제외재료 batch insert
  • 에러는 throw new CustomException(ErrorCode.XXX) — BusinessException 아님, 실제 클래스명 확인 필요
  • Spring 싱글턴 빈이라 인스턴스 필드에 요청별 상태(주문번호 등) 담지 말고 지역 변수/record로만 전달

힌트 — UserPayService.approvePayment

  • 순서 중요: 주문 존재(ORDER_NOT_FOUND) → 이미 승인됐는지(PAYMENT_ALREADY_APPROVED) → 금액 일치(ORDER_AMOUNT_MISMATCH) → 결제수단 활성(PAYMENT_METHOD_DISABLED), 이 순서로 막아야 함
  • 금액은 요청값이 아니라 orders.total_price(주문 생성 시 서버가 계산한 값)와 비교 — 결제 단계에서 또 클라이언트를 믿지 않음
  • insertPayment는 @Param 여러 개 대신 파라미터 객체(PaymentInsertParams)로 받아야 useGeneratedKeys가 생성된 id를 돌려받을 수 있음
  • 승인 후 getPaymentResult로 재조회해서 응답 조립 — 계산값을 직접 조립하지 않음 (waitingOrderCount가 실시간 값이라 뷰를 다시 읽어야 함)
확인 필요실제 코드 확인함

실제 프로젝트엔 BusinessException이 없고 com.asak.common.exception.CustomException(ErrorCode)가 있습니다. 던질 때 throw new CustomException(ErrorCode.MENU_NOT_FOUND) 형태로 씁니다.
ErrorCode에는 MENU_NOT_FOUND, MENU_SOLD_OUT, INVALID_OPTION_SELECTION, ORDER_NOT_FOUND, PAYMENT_METHOD_DISABLED, PAYMENT_ALREADY_APPROVED 등은 이미 있는데, OPTION_ITEM_SOLD_OUT, ORDER_AMOUNT_MISMATCH 이 2개는 없어서 추가해야 합니다.
